cmake_minimum_required(VERSION 3.22)
project(NullusPluginEngineIntegration LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This optional harness imports an existing Windows Debug engine build; use the root plugin test targets on other configurations.")
endif()
enable_testing()
get_filename_component(NLS_PLUGIN_REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(NLS_ENGINE_BUILD "${NLS_PLUGIN_REPO}/build" CACHE PATH "Build containing the real shared engine modules")
set(NLS_ENGINE_CONFIGURATION Debug CACHE STRING "Configuration of the imported modules")
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${NLS_PLUGIN_REPO}/ThirdParty/assimp/contrib/googletest" "${CMAKE_BINARY_DIR}/gtest" EXCLUDE_FROM_ALL)
foreach(module Core Base Math Platform Render Engine Scripting VFX Physics Animation Build Cook Plugins)
    file(GLOB_RECURSE candidates "${NLS_ENGINE_BUILD}/Runtime/*/${NLS_ENGINE_CONFIGURATION}/NLS_${module}d.lib")
    if(NOT candidates)
        message(FATAL_ERROR "Build NLS_${module} in ${NLS_ENGINE_BUILD} first")
    endif()
    list(GET candidates 0 library)
    add_library(NLS_${module} UNKNOWN IMPORTED)
    set_target_properties(NLS_${module} PROPERTIES IMPORTED_LOCATION "${library}")
endforeach()
set(_dotnet "${NLS_PLUGIN_REPO}/Tools/Dotnet/windows/x64/dotnet.exe")
set(_sample_output "${CMAKE_BINARY_DIR}/ManagedSample")
add_custom_target(NullusPluginManagedSample
    COMMAND "${_dotnet}" build "${NLS_PLUGIN_REPO}/Examples/Plugins/CSharpRuntime/Nullus.SamplePlugin.csproj"
        --configuration Debug --output "${_sample_output}" -p:DefineConstants=NULLUS_PLUGIN_ACCEPTANCE_FIXTURES --nologo
    VERBATIM)
add_executable(NullusPluginScriptTests ../ScriptAdapterTests.cpp)
target_compile_features(NullusPluginScriptTests PRIVATE cxx_std_20)
target_compile_definitions(NullusPluginScriptTests PRIVATE NLS_SHARED_LIB NLS_PLUGIN_REPO="${NLS_PLUGIN_REPO}"
    NLS_PLUGIN_DOTNET_ROOT="${NLS_PLUGIN_REPO}/Tools/Dotnet/windows/x64"
    NLS_PLUGIN_MANAGED_OUTPUT="${NLS_ENGINE_BUILD}/Managed/${NLS_ENGINE_CONFIGURATION}" NLS_PLUGIN_SAMPLE_OUTPUT="${_sample_output}")
target_include_directories(NullusPluginScriptTests PRIVATE "${NLS_PLUGIN_REPO}/Runtime" "${NLS_PLUGIN_REPO}/Runtime/Base"
    "${NLS_PLUGIN_REPO}/Runtime/Core" "${NLS_PLUGIN_REPO}/Runtime/Math" "${NLS_PLUGIN_REPO}/Runtime/Engine"
    "${NLS_PLUGIN_REPO}/Runtime/Rendering" "${NLS_PLUGIN_REPO}/ThirdParty")
foreach(_module Core Base Math Engine Rendering Scripting Physics Animation VFX UI)
    target_include_directories(NullusPluginScriptTests PRIVATE "${NLS_PLUGIN_REPO}/Runtime/${_module}/Gen")
endforeach()
target_link_libraries(NullusPluginScriptTests PRIVATE NLS_Scripting NLS_Engine NLS_Core NLS_Base NLS_Math NLS_Platform NLS_Render NLS_VFX NLS_Physics NLS_Animation gtest_main)
add_dependencies(NullusPluginScriptTests NullusPluginManagedSample)
file(GLOB _dlls "${NLS_ENGINE_BUILD}/runtime/${NLS_ENGINE_CONFIGURATION}/*.dll")
foreach(dll IN LISTS _dlls)
    add_custom_command(TARGET NullusPluginScriptTests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:NullusPluginScriptTests>" VERBATIM)
endforeach()
add_test(NAME NullusPluginScriptTests COMMAND NullusPluginScriptTests)
set_tests_properties(NullusPluginScriptTests PROPERTIES WORKING_DIRECTORY "${NLS_PLUGIN_REPO}" TIMEOUT 30)
add_executable(NullusPluginSerializationTests ../SerializationTests.cpp)
get_target_property(_includes NullusPluginScriptTests INCLUDE_DIRECTORIES)
target_include_directories(NullusPluginSerializationTests PRIVATE ${_includes})
target_compile_features(NullusPluginSerializationTests PRIVATE cxx_std_20)
target_compile_definitions(NullusPluginSerializationTests PRIVATE NLS_SHARED_LIB NLS_ECS_SHARED NLS_ENABLE_TEST_HOOKS=1 NLS_ENABLE_GAME_UI=1 NLS_ENABLE_PROFILING=1 NLS_HAS_VULKAN=0 NLS_HAS_METAL=0)
target_link_libraries(NullusPluginSerializationTests PRIVATE NLS_Engine NLS_Base NLS_Core NLS_Math NLS_Render NLS_Platform NLS_Physics NLS_Animation NLS_VFX gtest_main)
foreach(dll IN LISTS _dlls)
    add_custom_command(TARGET NullusPluginSerializationTests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:NullusPluginSerializationTests>" VERBATIM)
endforeach()
add_test(NAME NullusPluginSerializationTests COMMAND NullusPluginSerializationTests)
add_executable(NullusPluginAssetTests ../AssetCookTests.cpp)
target_include_directories(NullusPluginAssetTests PRIVATE ${_includes} "${NLS_PLUGIN_REPO}/Runtime/Platform"
    "${NLS_PLUGIN_REPO}/Runtime/Build" "${NLS_PLUGIN_REPO}/Runtime/Cook" "${NLS_PLUGIN_REPO}/Runtime/Plugins")
target_compile_features(NullusPluginAssetTests PRIVATE cxx_std_20)
target_compile_definitions(NullusPluginAssetTests PRIVATE NLS_SHARED_LIB)
target_link_libraries(NullusPluginAssetTests PRIVATE NLS_Build NLS_Cook NLS_Plugins NLS_Core NLS_Base gtest_main)
foreach(dll IN LISTS _dlls)
    add_custom_command(TARGET NullusPluginAssetTests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:NullusPluginAssetTests>" VERBATIM)
endforeach()
add_test(NAME NullusPluginAssetTests COMMAND NullusPluginAssetTests)
add_executable(NullusPluginManagedBuildTests ../ManagedBuildTests.cpp
    "${NLS_PLUGIN_REPO}/Project/Editor/Scripting/ManagedScriptBuildService.cpp")
target_include_directories(NullusPluginManagedBuildTests PRIVATE ${_includes}
    "${NLS_PLUGIN_REPO}/Project/Editor" "${NLS_PLUGIN_REPO}/Runtime/Platform"
    "${NLS_PLUGIN_REPO}/Runtime/Scripting" "${NLS_PLUGIN_REPO}/ThirdParty/Tracy/public")
target_compile_features(NullusPluginManagedBuildTests PRIVATE cxx_std_20)
target_compile_definitions(NullusPluginManagedBuildTests PRIVATE NLS_SHARED_LIB WITH_PROFILING=0
    NLS_PLUGIN_DOTNET_EXECUTABLE="${_dotnet}")
target_link_libraries(NullusPluginManagedBuildTests PRIVATE NLS_Scripting NLS_Core NLS_Platform NLS_Base NLS_Math gtest_main)
foreach(dll IN LISTS _dlls)
    add_custom_command(TARGET NullusPluginManagedBuildTests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:NullusPluginManagedBuildTests>" VERBATIM)
endforeach()
add_test(NAME NullusPluginManagedBuildTests COMMAND NullusPluginManagedBuildTests)
set_tests_properties(NullusPluginManagedBuildTests PROPERTIES TIMEOUT 240)
